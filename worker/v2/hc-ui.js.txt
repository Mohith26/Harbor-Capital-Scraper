  /* ------------------------------------------------------------------
     hcSplit(container, opts) \u2014 drag handle between two panes.
     The CSS of the container must size the controlled pane from a CSS
     variable, e.g. grid-template-columns: minmax(0,1fr) var(--map-w, 46%).
     opts: pane (el|selector, the pane whose size is controlled)
           dir 'x' (columns) | 'y' (rows)
           edge 'start' (handle on the pane's left/top edge, e.g. a right-hand map)
                | 'end' (handle on the pane's right/bottom edge, e.g. a left rail)
           varName CSS variable set on the container (default --hc-split)
           min, max (px for the pane), minOther (px kept for the other side)
           key localStorage key (persist), onResize(px)
     Returns { set(px), reset(), max(), min(), size() }.
     Leaflet maps and Chart.js charts pick up the change through the
     window resize event that is dispatched on every move.
     ------------------------------------------------------------------ */
  function hcSplit(container, opts) {

    if (!container) return null;

      } else {

  function isEmpty(f, v) {
    if (v == null || v === '') return true;
    if (f.type === 'multi') return !Array.isArray(v) || !v.length;
    if (f.type === 'range') return (v.min == null || v.min === '') && (v.max == null || v.max === '');
    if (f.type === 'daterange') return !v.from && !v.to;
    return false;
  }
  function clean(schema, value) {

    return out;
  }
  const fmtN = (n) => (n == null || n === '' ? '' : Number(n).toLocaleString('en-US'));
  function describe(f, v) {
    const u = f.unit ? ' ' + f.unit : '';

      const a = v.min !== '' && v.min != null, b = v.max !== '' && v.max != null;
      const fm = f.fmt || fmtN;
      const r = a && b ? `${fm(v.min)}\u2013${fm(v.max)}${u}` : a ? `\u2265 ${fm(v.min)}${u}` : `\u2264 ${fm(v.max)}${u}`;
      return `${f.short || f.label} ${r}${v.unk ? ' (+ unknown)' : ''}`;
    }
    if (f.type === 'multi') return `${f.short || f.label}: ${v.length > 2 ? v.slice(0, 2).join(', ') + ' +' + (v.length - 2) : v.join(', ')}`;
    if (f.type === 'tri') return `${f.short || f.label}: ${v === 'yes' ? 'Yes' : 'No'}`;
    if (f.type === 'daterange') return `${f.short || f.label} ${v.from || '\u2026'} \u2192 ${v.to || '\u2026'}`;
    if (f.type === 'select') { const o = (typeof f.options === 'function' ? f.options() : f.options || []).find((x) => String(x.v ?? x) === String(v)); return `${f.short || f.label}: ${o?.l ?? o?.v ?? v}`; }
    return `${f.short || f.label}: \u201c${v}\u201d`;
  }
  function hcFilterMatch(row, schema, value) {
    for (const s of schema) for (const f of s.fields) {
      const v = value[f.id]; if (isEmpty(f, v) || !f.get || f.server) continue;
      const x = f.get(row);
      if (f.type === 'range') {
        const n = x == null || x === '' ? null : Number(x);
        if (n == null || Number.isNaN(n)) { if (!v.unk) return false; continue; }
        if (v.min !== '' && v.min != null && n < Number(v.min)) return false;
        if (v.max !== '' && v.max != null && n > Number(v.max)) return false;
      } else if (f.type === 'multi') {
        const xs = (Array.isArray(x) ? x : [x]).filter((y) => y != null && y !== '').map(String);
        if (!xs.some((y) => v.includes(y))) return false;
      } else if (f.type === 'tri') {
        const t = x === true || x === 1 || x === '1' || /^(y|yes|true)$/i.test(String(x ?? ''));
        if ((v === 'yes') !== t) return false;
      } else if (f.type === 'text') {
        if (!String(x ?? '').toLowerCase().includes(String(v).toLowerCase())) return false;
      } else if (f.type === 'daterange') {
        const d = String(x ?? '').slice(0, 10); if (!d) return false;
        if (v.from && d < v.from) return false; if (v.to && d > v.to) return false;
      } else if (f.type === 'select') {
        if (String(x ?? '') !== String(v)) return false;
      }
    }
    return true;
  }

    el.innerHTML = fields.map((f) => `<span class="hc-fchip">${esc(describe(f, v[f.id]))}<button type="button" data-id="${esc(f.id)}" title="Remove">\u2715</button></span>`).join('')
      + (fields.length > 1 ? '<button type="button" class="clr">Clear all</button>' : '');
    el.querySelectorAll('.hc-fchip button').forEach((b) => (b.onclick = () => onRemove(b.dataset.id)));
    const cl = el.querySelector('.clr'); if (cl) cl.onclick = onClear;
  }

  /* compact URL-safe encoding for putting filters in the hash/query */
  const hcFilterEncode = (v) => (v && Object.keys(v).length ? encodeURIComponent(JSON.stringify(v)) : '');
  const hcFilterDecode = (s) => { try { return s ? JSON.parse(decodeURIComponent(s)) : {}; } catch { return {}; } };

  Object.assign(window, { hcSplit, hcColResize, hcFilters, hcFilterChips, hcFilterMatch, hcFilterEncode, hcFilterDecode, hcFilterDescribe: describe, hcStore: store });
})();
